cmake_minimum_required(VERSION 3.16)
project(CONFINDInstalledConsumer LANGUAGES CXX)
find_package(CONFIND 2.0 CONFIG REQUIRED)
get_target_property(options CONFIND::CONFIND INTERFACE_COMPILE_OPTIONS)
if(options MATCHES "ffp-contract")
  message(FATAL_ERROR "CONFIND must not impose its private FP policy on consumers")
endif()
add_executable(consumer main.cpp)
target_link_libraries(consumer PRIVATE CONFIND::CONFIND)
set_target_properties(consumer PROPERTIES CXX_EXTENSIONS OFF)
enable_testing()
add_test(NAME consumer_runs COMMAND consumer)
