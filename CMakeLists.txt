cmake_minimum_required(VERSION 3.16)
project(CONFIND VERSION 2.0.0 LANGUAGES CXX)
include(GNUInstallDirs)
find_package(Zaki 2.0 CONFIG REQUIRED)
find_package(Threads REQUIRED)
add_library(CONFIND STATIC source/Base.cpp source/Cell.cpp source/Cont2D.cpp source/ContourFinder.cpp)
add_library(CONFIND::CONFIND ALIAS CONFIND)
target_compile_features(CONFIND PUBLIC cxx_std_17)
set_target_properties(CONFIND PROPERTIES CXX_EXTENSIONS OFF)
target_include_directories(CONFIND PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
target_link_libraries(CONFIND PUBLIC Zaki::Zaki Threads::Threads)
if(CMAKE_CXX_COMPILER_ID MATCHES "^(AppleClang|Clang|GNU)$")
  target_compile_options(CONFIND PRIVATE -ffp-contract=off)
endif()
if(CMAKE_SOURCE_DIR STREQUAL PROJECT_SOURCE_DIR)
  set(CONFIND_TESTS_DEFAULT ON)
else()
  set(CONFIND_TESTS_DEFAULT OFF)
endif()
option(CONFIND_BUILD_TESTS "Build CONFIND qualification tests" ${CONFIND_TESTS_DEFAULT})
if(CONFIND_BUILD_TESTS)
  enable_testing()
  add_subdirectory(tests)
endif()
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
set_target_properties(CONFIND PROPERTIES VERSION ${PROJECT_VERSION} POSITION_INDEPENDENT_CODE ON)
install(TARGETS CONFIND EXPORT CONFINDTargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(DIRECTORY include/Confind DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(EXPORT CONFINDTargets FILE CONFINDTargets.cmake NAMESPACE CONFIND:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/CONFIND)
configure_package_config_file(cmake/CONFINDConfig.cmake.in ${CMAKE_CURRENT_BINARY_DIR}/CONFINDConfig.cmake
  INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/CONFIND)
write_basic_package_version_file(${CMAKE_CURRENT_BINARY_DIR}/CONFINDConfigVersion.cmake
  VERSION ${PROJECT_VERSION} COMPATIBILITY SameMajorVersion)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/CONFINDConfig.cmake ${CMAKE_CURRENT_BINARY_DIR}/CONFINDConfigVersion.cmake
  DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/CONFIND)
install(FILES legal/LICENSE DESTINATION ${CMAKE_INSTALL_DATAROOTDIR}/CONFIND)
