for mode in ['small','stellar']:
 for repeat in [1,2]:run([t/'m1-oracle',t/f'oracle-{mode}-{repeat}.tsv',mode,0,0],f'oracle-{mode}-{repeat}.log')

 run(['cmake','--build',b,'-j','6'],f'start-{typ}-build.log')

  run([b/'tests/confind_m1_characterize',t/f'start-{typ}-{mode}.tsv',mode,0,0],f'start-{typ}-{mode}.log')
  ref=(t/f'oracle-{mode}-1.tsv').read_bytes();got=(t/f'start-{typ}-{mode}.tsv').read_bytes()
